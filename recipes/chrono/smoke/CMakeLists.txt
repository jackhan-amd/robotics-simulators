cmake_minimum_required(VERSION 3.24)
project(robotics_fsi_smoke LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
option(STUDY_WITH_DEM "Build small DEM probe too" OFF)
if(STUDY_WITH_DEM)
  find_package(Chrono REQUIRED COMPONENTS FSI_SPH DEM CONFIG)
else()
  find_package(Chrono REQUIRED COMPONENTS FSI_SPH CONFIG)
endif()
add_executable(chrono_fsi_smoke fsi_sph.cpp)
target_link_libraries(chrono_fsi_smoke PRIVATE ${CHRONO_TARGETS})
if(STUDY_BACKEND STREQUAL "hip")
  find_package(hip REQUIRED CONFIG)
  target_compile_definitions(chrono_fsi_smoke PRIVATE CHRONO_USE_HIP)
  target_link_libraries(chrono_fsi_smoke PRIVATE hip::host)
elseif(STUDY_BACKEND STREQUAL "cuda")
  find_package(CUDAToolkit REQUIRED)
  target_compile_definitions(chrono_fsi_smoke PRIVATE CHRONO_USE_CUDA)
  target_link_libraries(chrono_fsi_smoke PRIVATE CUDA::cudart)
else()
  message(FATAL_ERROR "Set STUDY_BACKEND=hip or cuda")
endif()

if(STUDY_WITH_DEM)
  add_executable(chrono_dem_smoke dem.cpp)
  target_link_libraries(chrono_dem_smoke PRIVATE ${CHRONO_TARGETS})
  if(STUDY_BACKEND STREQUAL "hip")
    target_compile_definitions(chrono_dem_smoke PRIVATE CHRONO_USE_HIP)
    target_link_libraries(chrono_dem_smoke PRIVATE hip::host)
  else()
    target_compile_definitions(chrono_dem_smoke PRIVATE CHRONO_USE_CUDA)
    target_link_libraries(chrono_dem_smoke PRIVATE CUDA::cudart)
  endif()
endif()
